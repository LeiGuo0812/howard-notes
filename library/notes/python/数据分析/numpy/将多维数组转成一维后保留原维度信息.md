#Python #Numpy

```python
import numpy as np

my_array = np.random.randn(70, 2, 5)

# 将三维数组转换为一维数组
flattened_array = my_array.reshape(-1)

# 创建一个空的二维字符串数组，每一列对应不同的维度信息
info_array = np.empty((flattened_array.size, 3), dtype=object)

# 使用numpy的ndenumerate遍历三维数组的每个元素和它们的索引
for (i, j, k), value in np.ndenumerate(my_array):
    index = np.ravel_multi_index((i, j, k), my_array.shape)
    info_array[index] = [f"试次-{i+1}", f"通道-{j+1}", f"频率-{k+1}"]

print(flattened_array)
print(info_array)

```
